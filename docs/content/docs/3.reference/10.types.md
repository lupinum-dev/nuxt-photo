## Photo data and validation

```ts [PhotoItem]
interface PhotoItem<TMeta extends object = Readonly<Record<string, unknown>>> {
  readonly id: string
  readonly src: string
  readonly thumbSrc?: string
  readonly placeholderSrc?: string
  readonly width?: number
  readonly height?: number
  readonly alt?: string
  readonly caption?: string
  readonly description?: string
  readonly srcset?: string
  readonly meta?: Readonly<TMeta>
}
```

`id` must be a stable non-empty string, unique within the collection. Dimensions
are optional in the input type, but must resolve to positive finite numbers
before rendering. Both may be omitted with a configured dimension resolver or
Nuxt folder data. Explicit partial dimensions still fail. `thumbSrc` supplies a
thumbnail URL; `src` supplies the lightbox image. `meta` retains application data.

```ts [Validation types]
type PhotoValidationIssueCode =
  | 'missing-id'
  | 'missing-src'
  | 'invalid-width'
  | 'invalid-height'
  | 'duplicate-id'
  | 'invalid-item'
  | 'invalid-optional-field'
  | 'invalid-meta'

type PhotoValidationIssue = {
  readonly code: PhotoValidationIssueCode
  readonly owner: string
  readonly index: number
  readonly id?: string
  readonly message: string
}

type InvalidPhotoPolicy = 'throw' | 'drop'

type InvalidPhotosEvent = {
  readonly owner: string
  readonly issues: readonly PhotoValidationIssue[]
  readonly rawPhotos: readonly unknown[]
}
```

`PhotoValidationError` is a runtime-exported class that extends `Error`, with
readonly `owner: string` and `issues: readonly PhotoValidationIssue[]`.
`InvalidPhotoPolicy` defaults to `throw` in components. `drop` skips invalid
photos and every occurrence of duplicate IDs; recipes emit `invalidPhotos`.

## Configuration types

```ts [PhotoConfig]
interface PhotoProvider {
  url(src: string, options: { width: number; quality?: number; format?: string }): string
  placeholder?(src: string): string | undefined
  srcset?(photo: PhotoItem, context: 'thumb' | 'slide'): string | undefined
}
type LightboxTool = 'download' | 'share' | 'fullscreen'
interface LightboxOptions {
  component?: Component
  transition?: LightboxTransitionOption
  navigation?: LightboxNavigationMode
  minZoom?: number
  history?: boolean
  deepLink?: boolean | string
  tools?: LightboxTool[]
}
interface PhotoConfig {
  /** Image URL resolver. Provider candidate sizing is handled by the image pipeline. */
  provider?: PhotoProvider
  /** Locale code, or per-key overrides on top of the detected locale. */
  labels?: PhotoLocale | Partial<PhotoLabels>
  lightbox?: LightboxOptions
  /** Throw on invalid data by default; collection recipes can opt into dropping invalid photos. */
  validation?: InvalidPhotoPolicy
  /** Resolve missing intrinsic dimensions before validating photo data. */
  dimensions?: (src: string) => { width: number; height: number } | undefined
}
```

`PhotoConfig` is the input to `createPhoto`. `PhotoLocale` is
`'en' | 'de' | 'fr' | 'es' | 'it' | 'nl' | 'pt' | 'ar' | 'he'`.
The [configuration reference](/docs/reference/configuration-and-labels) lists
all `PhotoLabels` fields, defaults, merge rules, and timing.

### `PhotoProvider`

`url` is required. It receives a source URL and the requested rendition width.
`placeholder` optionally returns a preview URL. `srcset` optionally returns
fixed candidates for the full photo and `thumb` or `slide` context. Providers
do not supply `sizes`.

## Gallery and lightbox controllers

```ts [Controllers]
interface GalleryHandle<TMeta extends object = Readonly<Record<string, unknown>>> {
  open(index?: number): Promise<void>
  openById(id: string): Promise<void>
  close(): Promise<void>
  readonly isOpen: boolean
  readonly activeId: string | null
  readonly activePhoto: PhotoItem<TMeta> | null
}

interface LightboxController<TMeta extends object = Readonly<Record<string, unknown>>> {
  readonly photos: ComputedRef<readonly PhotoItem<TMeta>[]>
  readonly count: ComputedRef<number>
  readonly activeId: ComputedRef<string | null>
  readonly activeIndex: ComputedRef<number>
  readonly activePhoto: ComputedRef<PhotoItem<TMeta> | null>
  readonly isOpen: ComputedRef<boolean>
  open(index?: number): Promise<void>
  openById(id: string): Promise<void>
  close(): Promise<void>
  next(): void
  prev(): void
  toggleZoom(): void
}
```

Every ready-made component exposes `GalleryHandle`. The handle's state is
unwrapped for template refs and returns null photo state while closed.
`PhotoCarousel` also exposes `scrollTo(index: number): void`, `next(): void`,
and `prev(): void` for its inline track. `useLightbox` returns the controller,
whose state fields are computed refs; active photo state may remain available
while closed.

## Layout and responsive types

```ts [AlbumLayout]
type RowsAlbumLayout = {
  type: 'rows'
  targetRowHeight?: ResponsiveParameter<number>
}

type ColumnsAlbumLayout = {
  type: 'columns'
  columns?: ResponsiveParameter<number>
}

type MasonryAlbumLayout = {
  type: 'masonry'
  columns?: ResponsiveParameter<number>
}

type AlbumLayout = RowsAlbumLayout | ColumnsAlbumLayout | MasonryAlbumLayout

type ResponsiveParameter<T = number> = T | ((containerWidth: number) => T)

type ResponsivePhotoSizes = {
  size: string
  sizes?: Array<{ viewport: string; size: string }>
}
```

Rows accept `targetRowHeight` (default `300`); columns and masonry accept
`columns` (default `3`). Only the corresponding variant accepts each option.
`ResponsivePhotoSizes` describes sizes with viewport conditions.

## Animation and carousel types

```ts [Lightbox options]
type LightboxTransitionOption =
  | 'flip' | 'fade' | 'auto' | 'none'
  | { mode: 'flip' | 'fade' | 'auto' | 'none'; autoThreshold?: number }
type LightboxNavigationMode = 'slide' | 'fade' | 'crossfade'

interface PhotoCarouselAutoplayOptions {
  readonly delayMs?: number
  readonly stopOnInteraction?: boolean
  readonly stopOnMouseEnter?: boolean
}
```

Autoplay defaults: `4000` milliseconds, stop on interaction, and keep playing
on mouse enter. The delay must be positive and finite.

`CarouselControl` is `'arrows' | 'dots' | 'counter' | 'thumbnails'`.
`PhotoUi<T>` is a partial record of string classes. `T` selects a recipe:

| Type argument | Keys |
| --- | --- |
| `'Photo'` | `root`, `img`, `caption` |
| `'PhotoAlbum'` | `root`, `item`, `img` |
| `'PhotoGroup'` | `root` |
| `'PhotoCarousel'` | `root`, `slide`, `img`, `thumb`, `caption`, `controls` |

Omitting `T` allows the union of all recipe keys.

## Public slot-prop types

```ts [Slot props]
interface LightboxControlsSlotProps<
  TMeta extends object = Readonly<Record<string, unknown>>,
> {
  activeIndex: number
  activePhoto: PhotoItem<TMeta> | null
  photos: readonly PhotoItem<TMeta>[]
  count: number
  isZoomedIn: boolean
  zoomAllowed: boolean
  controlsDisabled: boolean
  next: () => void
  prev: () => void
  close: () => Promise<void>
  toggleZoom: () => void
}

interface LightboxCaptionSlotProps<
  TMeta extends object = Readonly<Record<string, unknown>>,
> {
  photo: PhotoItem<TMeta> | null
  activeIndex: number
}

interface LightboxSlideSlotProps<TMeta extends object = Readonly<Record<string, unknown>>> {
  photo: PhotoItem<TMeta>
  index: number
  width: number
  height: number
}

interface LightboxViewportSlotProps<
  TMeta extends object = Readonly<Record<string, unknown>>,
> {
  photos: readonly PhotoItem<TMeta>[]
  viewportRef: Ref<HTMLElement | null | undefined>
  imageLoadFailed: boolean
  isSlideMounted: (index: number) => boolean
}


interface CarouselSlideSlotProps<TMeta extends object = Readonly<Record<string, unknown>>> {
  photo: PhotoItem<TMeta>
  index: number
  selected: boolean
  open: () => void | Promise<void>
}

interface CarouselThumbSlotProps<TMeta extends object = Readonly<Record<string, unknown>>> {
  photo: PhotoItem<TMeta>
  index: number
  selected: boolean
  goTo: (i: number) => void
}

interface CarouselCaptionSlotProps<
  TMeta extends object = Readonly<Record<string, unknown>>,
> {
  photo: PhotoItem<TMeta>
  index: number
  count: number
}

interface CarouselControlsSlotProps {
  goToPrev: () => void
  goToNext: () => void
  canGoToPrev: boolean
  canGoToNext: boolean
  selectedIndex: number
  snapCount: number
  goTo: (i: number) => void
}

interface CarouselDotsSlotProps {
  snaps: readonly number[]
  selectedIndex: number
  goTo: (i: number) => void
}
```

Slot indexes are zero-based. `selected` describes the inline carousel
selection. `LightboxSlideSlotProps.width` and `height` describe fitted frame
pixels. See [Lightbox](/docs/reference/lightbox) for its recipe slot names;
they differ from the lower-level caption and controls slots.

## Nuxt module type

```ts [NuxtPhotoOptions]
import type { NuxtPhotoOptions } from '@lupinum/nuxt-photo'
```

`NuxtPhotoOptions` contains exactly `localImages`, `autoImports`, `components`,
`css`, `lightbox`, `validation`, `provider`, and `labels`, all optional. Their
types and defaults are in the [configuration reference](/docs/reference/configuration-and-labels).
